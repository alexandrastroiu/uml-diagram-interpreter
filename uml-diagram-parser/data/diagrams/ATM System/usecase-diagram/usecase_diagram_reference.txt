    usecase "Withdraw Cash" as UC_Withdraw

    usecase "Maintenance" as UC_Maintenance
    usecase "Repair" as UC_Repair
    usecase "ATM Transaction" as UC_Transaction
    usecase "ATM Help" as UC_Help
    usecase "Customer Authentication" as UC_Auth
    usecase "Replenishing" as UC_Replenish
    usecase "Upgrades" as UC_Upgrade
    usecase "Diagnostics" as UC_Diagnostics
}
